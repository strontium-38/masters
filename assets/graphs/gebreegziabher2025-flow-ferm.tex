\documentclass{standalone}
\usepackage{alone}

\begin{document}
\begin{tikzpicture}[
  widebox/.style = {draw, rounded corners=2pt, minimum width=5cm,
                    minimum height=0.9cm, align=center, font=\small, fill=blue!10},
  procbox/.style = {draw, rounded corners=2pt, minimum width=3.2cm,
                    minimum height=0.7cm, align=center, font=\small, fill=blue!25},
  prodbox/.style = {draw, rounded corners=2pt, minimum width=3.6cm,
                    minimum height=0.8cm, align=center, font=\small, fill=green!20},
  arr/.style     = {-{Stealth[length=2.2mm]}, semithick},
  darr/.style    = {-{Stealth[length=2.2mm]}, semithick, dashed},
]

  % Central vertical chain
  \node[widebox] (com) {Carbohydrates, proteins, and fats\\(Complex organic matter)};
  \node[procbox, below=of com] (hyd) {Hydrolysis};
  \node[widebox, below=of hyd] (sol) {Amino acids, sugars, and fatty acids\\(Soluble organic molecules)};
  \node[procbox, below=of sol] (acd) {Acidogenesis};
  \node[widebox, below=of acd] (int) {Intermediates\\(Short chain fatty acids and alcohols)};
  \node[procbox, below=of int] (act) {Acetogenesis};

  \coordinate[below=of act] (b_act);

  % Side products
  \node[prodbox, left=of b_act, xshift=-0.6cm] (aa) {Acetic acid};
  \node[prodbox, right=of b_act, xshift=0.6cm] (hc) {\ch{H2} and \ch{CO2}};

  % Methanogenesis
  \node[procbox, minimum width=4.4cm, below=of b_act, yshift=-0.25cm] (mg) {Methanogenesis};

  % Final product
  \node[prodbox, minimum width=4.4cm, below=of mg] (final) {\ch{CH4} and \ch{CO2}};

  % Main chain arrows
  \draw[arr] (com) -- (hyd);
  \draw[arr] (hyd) -- (sol);
  \draw[arr] (sol) -- (acd);
  \draw[arr] (acd) -- (int);
  \draw[arr] (int) -- (act);

  % Side lines from soluble organics (bypass)
  \draw[arr] (sol.west) -| ($(aa.north west)!0.1!(aa.north east)$);
  \draw[arr] (sol.east) -| ($(hc.north west)!0.9!(hc.north east)$);

  % Side lines from Intermediates (bypass)
  \draw[arr] (int.west) -| ($(aa.north west)!0.3!(aa.north east)$);
  \draw[arr] (int.east) -| ($(hc.north west)!0.7!(hc.north east)$);

  % Acetogenesis to side products
  \draw[arr] (act) -| (aa.north);
  \draw[arr] (act) -| (hc.north);

  % Solid arrows into final product
  \draw[arr] ($(aa.south west)!0.2!(aa.south east)$) |- (final.west);
  \draw[arr] ($(hc.south east)!0.2!(hc.south west)$) |- (final.east);
  \draw[arr] (mg) -- (final);

  % Named start points on the bottom edge of aa / hc
  \coordinate (L) at ($(aa.south west)!0.2!(aa.south east)$);
  \coordinate (R) at ($(hc.south east)!0.2!(hc.south west)$);

  % Points on those vertical legs at mg's height
  % (L |- mg) = same x as L, same y as mg
  \coordinate (mL) at (L |- mg);
  \coordinate (mR) at (R |- mg);

  % Merge lines from mg into the vertical legs
  % Dashed arrows into methanogenesis
  \draw[semithick, dashed] (mg.west) -- (mL);
  \draw[semithick, dashed] (mg.east) -- (mR);

  % Labels above the merges
  \node[font=\scriptsize, above] at ($(mg.west)!0.5!(mL)$) {Acetoclastic};
  \node[font=\scriptsize, above] at ($(mg.east)!0.5!(mR)$) {Hydrogenotrophic};
\end{tikzpicture}
\end{document}
