% total-flow.tex
\documentclass[border=10pt]{standalone}
\usepackage{pgfplots}
\usepackage{pgfplotstable}
\pgfplotsset{compat=1.18}

% ---- USER SETTINGS ----
\def\startday{0}      % change to e.g. 10 to start from day 10
\def\stopday{97}    % change to e.g. 80 to stop at day 80 (large number = all data)

\begin{document}
% Read CSV with explicit comma separator
\pgfplotstableread[col sep=comma]{experimentos-day.csv}{\data}

\begin{tikzpicture}
\begin{axis}[
    width=15cm, height=10cm,
    xlabel={Day},
    ylabel={Flow (nML/d)},
    title={Flow Rate},
    xmin=\startday, xmax=\stopday,
    ymin=0,
    grid=major,
    xtick distance=10,
    minor xtick={1,2,...,\stopday},
    x tick label style={/pgf/number format/1000 sep=},
    restrict x to domain=\startday:\stopday,
]
\addplot[
    color=red,
    thick,
    mark=*,
    mark size=1pt,
] table[x index=0, y index=2] {\data};
\end{axis}
\end{tikzpicture}
\end{document}
